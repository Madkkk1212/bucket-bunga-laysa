  <script>
    const splash = document.getElementById('splash');
    const mainStage = document.getElementById('mainStage');

    // Lock scroll saat splash tampil
    document.body.style.overflow = 'hidden';

    // Fungsi buka
    function openInvitation() {
      splash.classList.add('hide');
      mainStage.classList.add('show');

      setTimeout(() => {
        document.body.style.overflow = 'auto';
      }, 800);
    }

    // Klik / tap di splash
    splash.addEventListener('click', openInvitation);

    // Support touch untuk mobile
    splash.addEventListener('touchend', (e) => {
      e.preventDefault();
      openInvitation();
    }, { passive: false });

    // Support keyboard (Enter / Space) untuk aksesibilitas
    splash.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        openInvitation();
      }
    });

    splash.setAttribute('tabindex', '0');
  </script>
